```nix
# flake.nix - Declarative development environment
{
  description = "Paper Dispatch - Sleek personal publishing platform";

  inputs = {
    nixpkgs.url = "github:nixos/nixpkgs/nixos-unstable";
  };

  outputs = { self, nixpkgs }: let
    systems = [ "x86_64-linux" "aarch64-linux" ];
    forEachSystem = f: nixpkgs.lib.genAttrs systems (sys: f nixpkgs.legacyPackages.${sys});
  in {
    devShells = forEachSystem (pkgs: {
      default = pkgs.mkShell {
        packages = with pkgs; [ bun nodejs_22 gh git ];
      };
    });
  };
}
```

And here is a TypeScript snippet with arrow functions and type definitions:

```typescript
// src/lib/posts.ts - Fast local MDX reader
import fs from 'fs';
import path from 'path';
import matter from 'gray-matter';
import readingTime from 'reading-time';

export interface PostMeta {
  title: string;
  slug: string;
  date: string;
  category?: string;
  tags?: string[];
  readingTime: string;
}

export function getPostBySlug(slug: string): { meta: PostMeta; content: string } {
  const filePath = path.join(process.cwd(), 'content', 'posts', `${slug}.mdx`);
  const raw = fs.readFileSync(filePath, 'utf-8');
  const { data, content } = matter(raw);
  
  return {
    meta: { ...data, slug, readingTime: readingTime(content).text } as PostMeta,
    content,
  };
}
```
